```python
def start_recording_gesture(self):
    if not self.current_gesture_name:
        messagebox.showwarning("Ошибка", "Введите название жеста!")
        return
    
    self.is_recording = True
    self.recording_frames = []
    self.recording_start_time = time.time()

def process_video(self):
    # В цикле обработки видео
    if self.is_recording:
        hands_data, _ = self.gesture_detector.detect(frame)
        landmarks = hands_data.get('left') or hands_data.get('right')
        
        if landmarks is not None:
            self.recording_frames.append(landmarks)
        
        # Остановка через 3 секунды
        if time.time() - self.recording_start_time > 3.0:
            self.stop_recording_gesture()

def stop_recording_gesture(self):
    self.is_recording = False
    
    # Сохраняем каждый кадр как отдельный пример
    for landmarks in self.recording_frames:
        self.data_collector.add_sample(
            self.current_gesture_name,
            landmarks
        )
    
    # Обновляем счётчик в GUI
    self.update_gesture_list()
```
