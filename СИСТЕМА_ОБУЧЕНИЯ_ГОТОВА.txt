┗━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━┛

Input (63 признака - координаты 21 точки руки)
   ↓
Input Projection: Linear(63 → 256) + BatchNorm + ReLU + Dropout
   ↓
ResidualBlock 1: (256 → 512)
   ├─→ FC → BN → ReLU → Dropout → FC → BN ─→┐
   └──────────────────────────────────────→ ADD → ReLU
   ↓
ResidualBlock 2: (512 → 512)
   ├─→ FC → BN → ReLU → Dropout → FC → BN ─→┐
   └──────────────────────────────────────→ ADD → ReLU
   ↓
Attention Layer: (512)
   Input → FC(512→128) → ReLU → FC(128→512) → Sigmoid
          ↓
   Input * Attention_Weights
   ↓
ResidualBlock 3: (512 → 512)
   ├─→ FC → BN → ReLU → Dropout → FC → BN ─→┐
   └──────────────────────────────────────→ ADD → ReLU
   ↓
ResidualBlock 4: (512 → 256)
   ├─→ FC → BN → ReLU → Dropout → FC → BN ─→┐
   └──────────────────────────────────────→ ADD → ReLU
   ↓
Classifier: Linear(256 → 128) + BN + ReLU + Dropout + Linear(128 → num_classes)
   ↓
Output (вероятности классов)

```python
from training.training_module import DataCollector, ModelTrainer
from model.advanced_gesture_model import AdvancedGestureClassifier

# 1. Создать сборщик данных
collector = DataCollector()

# 2. Добавить примеры жестов
for i in range(20):
    landmarks = get_hand_landmarks()  # получить из камеры
    collector.add_sample("Peace", landmarks)

for i in range(20):
    landmarks = get_hand_landmarks()
    collector.add_sample("OK", landmarks)

# 3. Подготовить данные
X_train, X_test, y_train, y_test, gesture_classes = \
    collector.prepare_training_data()

# 4. Создать модель
model = AdvancedGestureClassifier(
    input_size=63,
    num_classes=len(gesture_classes)
)

# 5. Создать trainer
trainer = ModelTrainer(model)

# 6. Добавить callback для прогресса
def on_progress(event, data):
    if event == 'epoch_end':
        print(f"Эпоха {data['epoch']}: "
              f"Train Acc = {data['train_acc']:.2f}%, "
              f"Val Acc = {data['val_acc']:.2f}%")

trainer.add_callback(on_progress)

# 7. Обучить модель
from torch.utils.data import DataLoader
from training.training_module import GestureDataset

train_dataset = GestureDataset(X_train, y_train)
test_dataset = GestureDataset(X_test, y_test)

train_loader = DataLoader(train_dataset, batch_size=16, shuffle=True)
test_loader = DataLoader(test_dataset, batch_size=16)

trainer.train(train_loader, test_loader, epochs=50)

# 8. Сохранить модель
recognizer = AdvancedGestureRecognizer()
recognizer.model = model
recognizer.gesture_classes = gesture_classes
recognizer.save_model("models/my_advanced_model.pth")
```
